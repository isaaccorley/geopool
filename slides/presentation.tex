\begin{frame}{Pooling is the bottleneck}
  \begin{columns}[c, onlytextwidth]
    \column{0.42\textwidth}
    \begin{itemize}
      \item Pixel-level GFMs output one embedding \emph{per pixel}.
      \item Downstream labels live at the \textbf{patch / region} level.
      \item A EuroSAT patch contains \textbf{4,096} embeddings that must be collapsed to one vector.
    \end{itemize}

    \vspace{0.5em}
    \begin{alertblock}{Why this matters}
      Mean pooling can lose \textbf{8.8 pp} under geographic shift.
    \end{alertblock}

    \column{0.54\textwidth}
    \centering
    \includegraphics[width=\linewidth]{figures/patch_vs_pixel.pdf}
  \end{columns}
\end{frame}

\begin{frame}{Mean misses within-patch structure}
  \begin{columns}[c, onlytextwidth]
    \column{0.24\textwidth}
    \begin{itemize}
      \item Homogeneous classes cluster tightly.
      \item Heterogeneous classes show \alert{broad distributions}.
      \item Mean pooling keeps the center and discards the spread.
    \end{itemize}

    \column{0.72\textwidth}
    \centering
    \includegraphics[width=\linewidth]{figures/heterogeneity.pdf}
  \end{columns}

  \vspace{0.2em}
  {\small A residential patch and an industrial patch can share a similar mean while differing sharply in variance.}
\end{frame}

\begin{frame}{Distributional statistics improve spatial generalization}
  \begin{columns}[c, onlytextwidth]
    \column{0.38\textwidth}
    \begin{block}{Spatial linear probe average}
      \renewcommand{\arraystretch}{1.25}
      \centering
      \begin{tabular}{lr}
        Mean & 87.3 \\
        Mean+Max & 91.8 \\
        Covariance & 92.8 \\
        \textbf{Stats} & \textbf{93.0} \\
      \end{tabular}
    \end{block}

    \begin{alertblock}{Key result}
      \textbf{Stats} gains \textbf{+5.7 pp} over Mean on spatial splits.\\[0.3em]
      On random splits, the gap shrinks to \textbf{$\sim$1 pp}.
    \end{alertblock}

    {\small Spatial evaluation is the honest test.}

    \column{0.58\textwidth}
    \centering
    \includegraphics[width=\linewidth]{../paper/figures/random_vs_spatial.pdf}

    {\footnotesize Closer to diagonal = smaller random$\to$spatial gap}
  \end{columns}
\end{frame}

\begin{frame}{Accuracy and robustness are aligned}
  \begin{columns}[c, onlytextwidth]
    \column{0.52\textwidth}
    \centering
    \includegraphics[width=\linewidth]{../paper/figures/random_vs_spatial_linear.pdf}

    \column{0.44\textwidth}
    \begin{block}{Average gap (linear probe)}
      \renewcommand{\arraystretch}{1.3}
      \centering
      \begin{tabular}{lr}
        Mean & \textbf{8.8 pp} \\
        Mean+Max & 4.6 pp \\
        Covariance & 4.3 pp \\
        Stats & 3.8 pp \\
      \end{tabular}
    \end{block}

    \vspace{0.35em}
    \begin{itemize}\small
      \item No accuracy--robustness tradeoff.
      \item Best spatial methods also have the smallest gaps.
      \item \textbf{Stats}, \textbf{Covariance}, and \textbf{Mean+Max} all dominate Mean.
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Takeaways and scope}
  \begin{columns}[T, onlytextwidth]
    \column{0.56\textwidth}
    \begin{block}{Takeaways}
      \begin{itemize}
        \item Pooling is \textbf{not} a benign preprocessing step for fixed pixel-level embedding products.
        \item Simple distributional summaries materially improve \textbf{spatial} generalization.
        \item \textbf{Stats} is the best default; \textbf{Covariance} is strongest when dimensionality is less constrained.
      \end{itemize}
    \end{block}

    \column{0.38\textwidth}
    \begin{block}{Scope}
      \begin{itemize}
        \item Single benchmark: EuroSAT
        \item Three released embedding products
        \item Post-hoc pooling only; no encoder access
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}
